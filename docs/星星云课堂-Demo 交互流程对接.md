```javascript
// 伪代码：自动保存逻辑
function saveProgress(questionId, answer) {
  localStorage.setItem(`assessment_${userId}_${questionId}`, JSON.stringify({
    answer: answer,
    timestamp: Date.now(),
    version: 'v1.0'
  }));
  
  // 每 5 题同步到服务器
  if (currentQuestionIndex % 5 === 0) {
    syncToServer();
  }
}

// 断点续答
function resumeAssessment() {
  const savedProgress = loadProgress();
  if (savedProgress) {
    showResumePrompt();  // "检测到未完成的评估，是否继续？"
  }
}
```

## 三、数据流转设计

### 3.1 数据模型

```typescript
// TypeScript 类型定义

interface User {
  id: string;
  name: string;  // 孩子姓名
  age: number;
  registrationDate: string;
  hasCompletedAssessment: boolean;
  lastAssessmentDate?: string;
}

interface Assessment {
  id: string;
  userId: string;
  completedAt: string;
  duration: number;  // 分钟
  questions: AssessmentQuestion[];
  overall: OverallResult;
  dimensions: DimensionResult[];
}

interface AssessmentQuestion {
  id: string;
  number: number;
  dimension: string;
  type: 'parent_observation' | 'interactive';
  answer: string;
  score: number;
}

interface OverallResult {
  level: 'L1' | 'L2' | 'L3';
  levelName: string;
  totalScore: number;
  maxScore: number;
}

interface DimensionResult {
  id: string;
  name: string;
  score: number;
  maxScore: number;
  percentage: number;
  level: 'L1' | 'L2' | 'L3';
  subDimensions: SubDimension[];
  suggestions: string[];
}

interface Course {
  id: string;
  name: string;
  dimension: string;
  duration: number;
  level: 'L1' | 'L2' | 'L3';
  videoUrl: string;
  completed: boolean;
}
```

```
开始学习 (30%)
```

### 4.2 关键指标

| 指标 | 定义 | 目标值 |
|------|------|--------|
| 评估渗透率 | 完成评估用户/总用户 | >50% |
| 评估完成率 | 完成评估/开始评估 | >80% |
| 评估平圴用时 | 完成评估的平圴时长 | 15-20 分钟 |
| 课程转化率 | 评估后开始学习/完成评估 | >70% |
| 复评率 | 3 个月内复评用户/到期用户 | >40% |
